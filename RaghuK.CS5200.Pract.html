# Function to create and populate the airports table dataframe
create_airports_df <- function(bds.raw) {
  
  unique_airports <- unique(bds.raw[, c("dep_airport", "origin_state")])
  
  airports_df <- data.frame(
    aid = seq_len(nrow(unique_airports)),
    airportName = unique_airports$dep_airport,
    airportState = unique_airports$origin_state,
    airportCode = "ZZZ",
    stringsAsFactors = FALSE
  )
  
  return(airports_df)
}

airports_df <- create_airports_df(bds.raw)
dbWriteTable(con, name = "airports", value = airports_df, row.names = FALSE, append = TRUE)

# Function to create and populate the flights table dataframe
process_flights_df <- function(bds.raw, airports_df) {
  
  bds.raw$airport_key <- paste(bds.raw$dep_airport, bds.raw$origin_state, sep = "_")
  airports_df$airport_key <- paste(airports_df$airportName, airports_df$airportState, sep = "_")
  
  match_indices <- match(bds.raw$airport_key, airports_df$airport_key)
  bds.raw$originAirport <- airports_df$aid[match_indices]
  
 # bds.raw$airline[is.na(bds.raw$airline) | bds.raw$airline == ""] <- "Unknown"
  bds.raw$aircraftType <- bds.raw$aircraft
 # bds.raw$isHeavy <- ifelse(bds.raw$heavy_flag == "Yes", TRUE, FALSE)
  bds.raw$isHeavy <- ifelse(
  bds.raw$heavy_flag == "Yes", 
  1, 
  ifelse(bds.raw$heavy_flag == "unknown" | bds.raw$heavy_flag == "", NA, 0)
)
  
  bds.raw$date <- as.Date(bds.raw$flight_date, format = "%m/%d/%y")
  
  flights_df <- data.frame(
    fid = bds.raw$iid,
    date = bds.raw$date,
    originAirport = bds.raw$originAirport,
    airlineName = bds.raw$airline,
    aircraftType = bds.raw$aircraftType,
    isHeavy = as.integer(bds.raw$isHeavy),
    stringsAsFactors = FALSE
  )
  
  return(flights_df)
}

flights_df <- process_flights_df(bds.raw, airports_df)
dbWriteTable(con, name = "flights", value = flights_df, row.names = FALSE, append = TRUE)

# Function to create and populate the wildlife_strikes table dataframe
process_incidents_df <- function(bds.raw, conditions_df, wildlife_size_df, flights_df) {

  # Assign fid values from flights_df to bds.raw
  bds.raw$fid <- flights_df$fid

  # Map conditions to get cid values from conditions_df
  match_conditions <- match(bds.raw$sky_conditions, conditions_df$sky_condition)
  bds.raw$conditions <- conditions_df$cid[match_conditions]

  # Map wildlife size to get wlsizeid from wildlife_size_df
  match_sizes <- match(bds.raw$wildlife_size, wildlife_size_df$description)
  bds.raw$wlsize <- wildlife_size_df$wlsizeid[match_sizes]

  # Handle altitude, replacing invalid values with 0
  bds.raw$altitude_ft[is.na(bds.raw$altitude_ft) | bds.raw$altitude_ft == "" | bds.raw$altitude_ft < 0] <- 0
  bds.raw$altitude_ft <- gsub(",", "", bds.raw$altitude_ft)  # Remove commas if present
  bds.raw$altitude <- as.integer(bds.raw$altitude_ft)

  # Assign unique incident IDs (iid)
  bds.raw$iid <- seq_len(nrow(bds.raw))

  # Create incidents_df for wildlife_strikes table insertion
  incidents_df <- data.frame(
    iid = bds.raw$iid,
    fid = bds.raw$fid,
    wlsize = bds.raw$wlsize,
    impact = bds.raw$impact,
    altitude = bds.raw$altitude,
    conditions = bds.raw$conditions,
    stringsAsFactors = FALSE
  )

  return(incidents_df)
}

# Run the process and insert into the wildlife_strikes table
incidents_df <- process_incidents_df(bds.raw, conditions_df, wildlife_size_df, flights_df)
dbWriteTable(con, name = "wildlife_strikes", value = incidents_df, row.names = FALSE, append = TRUE)
